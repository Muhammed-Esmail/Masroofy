 12class HistoryManager:
 13    """
 14    A Facade service that orchestrates cycle and transaction data retrieval.
 15
 16    HistoryManager provides a high-level API for the rest of the application to 
 17    interact with historical data, shielding them from the underlying complexities 
 18    of the specific CRUD services.
 19
 20    Attributes:
 21        transaction_queryable (TransactionCRUD): Service for transaction data access.
 22        cycle_readable (CycleCRUD): Service for cycle data access and state management.
 23    """
 24    _instance = None
 25
 26    def __new__(cls, *args, **kwargs):
 27        """Ensures a Singleton instance of HistoryManager."""
 28        if cls._instance is None:
 29            cls._instance = super().__new__(cls)
 30        return cls._instance
 31    
 32    transaction_queryable: TransactionCRUD
 33    cycle_readable: CycleCRUD
 34
 35    def __init__(self):
 36        """Initializes the manager and connects the required service layers."""
 37        self.transaction_queryable = TransactionCRUD()
 38        self.cycle_readable = CycleCRUD()
 39
 40    def fetchCycleData(self, cycle_id: int):
 41        """
 42        Retrieves details for a specific cycle.
 43
 44        Args:
 45            cycle_id (int): The unique identifier of the cycle.
 46
 47        Returns:
 48            Cycle: The cycle data object.
 49        """
 50        return self.cycle_readable.fetchById(cycle_id)
 51    
 52    def fetchTransactionData(self, startDate: date, endDate: date, category_name: str, cycle_id: int):
 53        """
 54        Retrieves transactions based on multiple filtering criteria.
 55
 56        Args:
 57            startDate (date): Lower bound for transaction date.
 58            endDate (date): Upper bound for transaction date.
 59            category_name (str): Category filter.
 60            cycle_id (int): Cycle filter.
 61
 62        Returns:
 63            list[Transaction]: A list of transactions matching the criteria.
 64        """
 65        return self.transaction_queryable.fetchByFilters(startDate, endDate, category_name, cycle_id)
 66    
 67    def fetchFullHistory(self, currentOnly: bool = False):
 68        """
 69        Retrieves transaction history, optionally restricted to the active cycle.
 70
 71        Args:
 72            currentOnly (bool): If True, only fetches transactions for the active cycle. 
 73                               Defaults to False (all history).
 74
 75        Returns:
 76            list[Transaction]: The requested transaction history.
 77
 78        Raises:
 79            NoActiveCycle: If currentOnly is True but no cycle is active.
 80        """
 81        activeCycleID = self.getActiveCycleID()
 82        if activeCycleID is None:
 83            raise NoActiveCycle("[History Manager] No Active Cycle!")
 84        if currentOnly:
 85            return self.transaction_queryable.fetchByFilters(cycle_id=activeCycleID)
 86        else:
 87            return self.transaction_queryable.fetchByFilters()
 88
 89    def createNewCycle(self, newCycle: CycleModel) -> int:
 90        """
 91        Registers a new budget cycle.
 92
 93        Args:
 94            newCycle (CycleModel): The model data for the new cycle.
 95
 96        Returns:
 97            int: The ID of the newly created cycle.
 98        """
 99        return self.cycle_readable.create(newCycle)
100
101    def getActiveCycle(self) -> Optional[Cycle]:
102        """
103        Returns the currently active Cycle data object.
104
105        Returns:
106            Optional[Cycle]: The active cycle, or None if no cycle is set.
107        """
108        return self.cycle_readable.fetchActive()
109
110    def getActiveCycleID(self) -> Optional[int]:
111        """
112        A helper to quickly retrieve the ID of the active cycle.
113
114        Returns:
115            Optional[int]: The cycle ID, or None if no cycle is active.
116        """
117        currentCycle = self.getActiveCycle()
118        if currentCycle:
119            return currentCycle.id
120        return None
121    
122    def getTotalSpentInActiveCycle(self) -> Optional[int]:
123        """
124        Aggregates spending for the current active cycle.
125
126        Returns:
127            Optional[int]: The total sum spent, or 0 if no cycle is active 
128                           or no transactions exist.
129        """
130        currentCycleId = self.getActiveCycleID()
131        if currentCycleId:
132            return self.transaction_queryable.getTotalSpentInCycle(currentCycleId)
133        return 0
134
135    def setActive(self, id: int) -> None:
136        """
137        Sets a specific cycle as the active one in the system.
138
139        Args:
140            id (int): The ID of the cycle to activate.
141        """
142        self.cycle_readable.setActive(id)
143
144    def clearActive(self) -> None:
145        """Removes the active status from any currently active cycle."""
146        self.cycle_readable.clearActive()
147
148    def updateCycleData(self, cycleID: int, newData: CycleModel) -> None:
149        """
150        Updates the parameters of an existing cycle.
151
152        Args:
153            cycleID (int): The ID of the cycle to update.
154            newData (CycleModel): The new data values to persist.
155        """
156        self.cycle_readable.update(cycleID, newData)
